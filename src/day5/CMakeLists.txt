cmake_minimum_required(VERSION 3.24)
project(iot25_day5_student_demo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

foreach(name sensor api bridge consumer contract_test)
    add_executable(${name} ${name}.cpp)
    target_include_directories(${name} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/include")
    target_link_libraries(${name} PRIVATE Threads::Threads)
    if(WIN32)
        target_link_libraries(${name} PRIVATE ws2_32 crypt32)
    endif()
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

enable_testing()
add_test(NAME contract COMMAND contract_test)
