cmake_minimum_required(VERSION 3.24)
project(iot25_day4_data_formats LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_Declare(pugixml GIT_REPOSITORY https://github.com/zeux/pugixml.git GIT_TAG v1.15 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(json pugixml)
add_executable(json_demo json_demo.cpp)
target_link_libraries(json_demo PRIVATE nlohmann_json::nlohmann_json)
add_executable(xml_demo xml_demo.cpp)
target_link_libraries(xml_demo PRIVATE pugixml::pugixml)
foreach(target json_demo xml_demo)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
