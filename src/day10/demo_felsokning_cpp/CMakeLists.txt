cmake_minimum_required(VERSION 3.16)
project(iot25_day10_troubleshooting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(troubleshooting_server server.cpp)
add_executable(troubleshooting_client client.cpp)
if(MSVC)
  target_compile_options(troubleshooting_server PRIVATE /W4)
  target_compile_options(troubleshooting_client PRIVATE /W4)
else()
  target_compile_options(troubleshooting_server PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(troubleshooting_client PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(troubleshooting_server PRIVATE ws2_32)
  target_link_libraries(troubleshooting_client PRIVATE ws2_32)
endif()
